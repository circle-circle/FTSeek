    # P0 优化：检查例外情况

    is_high_confidence = confidence is not None and confidence > 70
    
    if is_overshoot:
        logger.info(f"✅ 例外：超跌反弹 (RSI={rsi:.1f}<20)，允许买入")

    if is_high_confidence:
        logger.info(f"✅ 例外：高置信度信号 (confidence={confidence:.1f}%>70%)，允许买入")
